static const int NET[19][2] = {
    {0, 1}, {2, 3}, {4, 5}, {6, 7}, {0, 2}, {1, 3}, {4, 6}, {5, 7}, {1, 2}, {5, 6},
    {0, 4}, {3, 7}, {1, 5}, {2, 6}, {1, 4}, {3, 6}, {2, 4}, {3, 5}, {3, 4},
};

for (long g = 0; g < n; g += 8) {
    long *v = a + g;
    for (int c = 0; c < 19; c++) {
        long lo = v[NET[c][0]], hi = v[NET[c][1]];
        if (lo > hi) {
            v[NET[c][0]] = hi;
            v[NET[c][1]] = lo;
        }
    }
}
